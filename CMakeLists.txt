cmake_minimum_required(VERSION 3.20)
project(AstraCore LANGUAGES CXX)
find_package(Threads REQUIRED)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(astra_core STATIC ${CORE_SOURCES})
target_include_directories(astra_core PUBLIC core)
target_compile_features(astra_core PUBLIC cxx_std_20)
target_link_libraries(astra_core PUBLIC Threads::Threads)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
add_executable(astra_tests ${TEST_SOURCES})
target_link_libraries(astra_tests PRIVATE astra_core)
enable_testing()
add_test(NAME astra_core COMMAND astra_tests)
file(GLOB DEMO_SOURCES CONFIGURE_DEPENDS demo/*.cpp)
add_executable(astra_demo ${DEMO_SOURCES})
target_link_libraries(astra_demo PRIVATE astra_core)
if(MINGW)
    target_link_options(astra_demo PRIVATE -municode)
endif()
add_test(NAME astra_demo COMMAND astra_demo --smoke)

if(WIN32)
    add_library(astra_tcp STATIC net/tcp_stream.cpp)
    target_link_libraries(astra_tcp PUBLIC astra_core ws2_32)
    file(GLOB TCP_TEST_SOURCES CONFIGURE_DEPENDS network_tests/*.cpp)
    add_executable(astra_tcp_tests ${TCP_TEST_SOURCES})
    target_link_libraries(astra_tcp_tests PRIVATE astra_tcp)
    add_test(NAME astra_tcp COMMAND astra_tcp_tests)
    set_tests_properties(astra_tcp PROPERTIES TIMEOUT 30)
endif()

option(ASTRA_SQLITE "Build the bundled SQLite adapter and integration runner" OFF)
if(ASTRA_SQLITE)
    enable_language(C)
    set(SQLITE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.tools/sqlite-amalgamation-3530400")
    if(NOT EXISTS "${SQLITE_SOURCE}/sqlite3.c")
        message(FATAL_ERROR "Run scripts/setup-sqlite.ps1 first.")
    endif()
    add_library(astra_sqlite_engine STATIC "${SQLITE_SOURCE}/sqlite3.c")
    target_include_directories(astra_sqlite_engine PUBLIC "${SQLITE_SOURCE}")
    target_compile_definitions(astra_sqlite_engine PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
    file(GLOB SQLITE_SOURCES CONFIGURE_DEPENDS storage/sqlite*.cpp)
    add_library(astra_sqlite STATIC ${SQLITE_SOURCES})
    target_include_directories(astra_sqlite PUBLIC storage)
    target_link_libraries(astra_sqlite PUBLIC astra_core astra_sqlite_engine)
    if(WIN32)
        file(GLOB GAME_SOURCES CONFIGURE_DEPENDS game/*.cpp)
        add_executable(astra_game ${GAME_SOURCES})
        set_target_properties(astra_game PROPERTIES OUTPUT_NAME astra-game)
        target_link_libraries(astra_game PRIVATE astra_sqlite)
        if(MINGW)
            target_link_options(astra_game PRIVATE -municode)
        endif()
    endif()
    add_executable(astra_saved_demo ${DEMO_SOURCES})
    target_compile_definitions(astra_saved_demo PRIVATE ASTRA_DEMO_SQLITE)
    target_link_libraries(astra_saved_demo PRIVATE astra_sqlite)
    add_executable(astra_storage_benchmark benchmark/main.cpp)
    target_link_libraries(astra_storage_benchmark PRIVATE astra_sqlite)
    if(MINGW)
        target_link_options(astra_saved_demo PRIVATE -municode)
        target_link_options(astra_storage_benchmark PRIVATE -municode)
    endif()
    file(GLOB SQLITE_TEST_SOURCES CONFIGURE_DEPENDS integration/sqlite*.cpp)
    add_executable(astra_sqlite_tests ${SQLITE_TEST_SOURCES})
    target_link_libraries(astra_sqlite_tests PRIVATE astra_sqlite)
    if(WIN32)
        target_sources(astra_sqlite_tests PRIVATE network_tests/loopback.cpp)
        target_link_libraries(astra_sqlite_tests PRIVATE astra_tcp)
        find_program(POWERSHELL NAMES pwsh powershell REQUIRED)
        add_test(NAME astra_sqlite COMMAND "${POWERSHELL}" -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-sqlite.ps1" -Executable $<TARGET_FILE:astra_sqlite_tests>)
        add_test(NAME astra_saved_demo COMMAND "${POWERSHELL}" -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-demo.ps1" -Executable $<TARGET_FILE:astra_saved_demo>)
        add_test(NAME astra_client_demo COMMAND "${POWERSHELL}" -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-demo.ps1" -Executable $<TARGET_FILE:astra_saved_demo> -ClientMode)
        add_test(NAME astra_survival COMMAND "${POWERSHELL}" -NoProfile -File
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-game.ps1" -Executable $<TARGET_FILE:astra_game>)
    endif()
endif()

option(ASTRA_ASSETS "Build the pinned MikkTSpace asset adapter" OFF)
if(ASTRA_ASSETS)
    enable_language(C)
    set(MIKK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.tools/mikktspace")
    if(NOT EXISTS "${MIKK_SOURCE}/mikktspace.c")
        message(FATAL_ERROR "Run scripts/build-assets.ps1 first to prepare pinned MikkTSpace sources.")
    endif()
    file(SHA256 "${MIKK_SOURCE}/mikktspace.c" MIKK_HASH)
    if(NOT MIKK_HASH STREQUAL "de87e74107df766ce68108801262bd8d53899414236b59810509a8fc2a51e288")
        message(FATAL_ERROR "MikkTSpace source checksum mismatch.")
    endif()
    add_library(astra_mikktspace STATIC "${MIKK_SOURCE}/mikktspace.c")
    target_include_directories(astra_mikktspace PUBLIC "${MIKK_SOURCE}")
    add_executable(astra_mikk tools/mikk.cpp)
    set_target_properties(astra_mikk PROPERTIES OUTPUT_NAME astra-mikk)
    target_compile_features(astra_mikk PRIVATE cxx_std_20)
    target_link_libraries(astra_mikk PRIVATE astra_mikktspace)
endif()
