cmake_minimum_required(VERSION 3.20)
project(AstraCore LANGUAGES CXX)
find_package(Threads REQUIRED)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(astra_core STATIC ${CORE_SOURCES})
target_include_directories(astra_core PUBLIC core)
target_compile_features(astra_core PUBLIC cxx_std_20)
target_link_libraries(astra_core PUBLIC Threads::Threads)
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
add_executable(astra_tests ${TEST_SOURCES})
target_link_libraries(astra_tests PRIVATE astra_core)
enable_testing()
add_test(NAME astra_core COMMAND astra_tests)
file(GLOB DEMO_SOURCES CONFIGURE_DEPENDS demo/*.cpp)
add_executable(astra_demo ${DEMO_SOURCES})
target_link_libraries(astra_demo PRIVATE astra_core)
add_test(NAME astra_demo COMMAND astra_demo --smoke)
